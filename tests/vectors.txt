#   model <name>                       select a model by cs_model_by_name
#   none <name>                        cs_model_by_name finds nothing
#   usb <vid> <pid> <model name>|none  cs_model_by_usb
#   port <name> <roles>                cs_port_roles for the selected model: "none" or
#                                      keys, controls, display joined by "+"
#   cable <index> <roles>              cs_port_roles_at
#   decode <ms> <status> <d1> <d2> -> <kind> <index> <value> <delta>
#   connect|disconnect -> <bytes>
#   text <line1> <line2> -> <bytes>
#   value knob|fader <value> <line1> <line2> -> <bytes>
#   pad <pad> <r> <g> <b> -> <bytes>
#
# Strings are in double quotes. <bytes> is hex, or "none" for a 0 return.
# A decoder lives from one "model" line to the next.
